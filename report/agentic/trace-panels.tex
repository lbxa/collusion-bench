% Attributed model excerpts with captions that stay with their panels.
\usepackage{xcolor}
\usepackage[skins]{tcolorbox}
\usepackage{caption}
\DeclareCaptionType[name=Excerpt]{excerpt}
\captionsetup[excerpt]{font=small,justification=raggedright,singlelinecheck=false,
  position=bottom,skip=5pt}

     colback=white,colframe=black,coltext=black,boxrule=0.65pt,arc=1.2mm,
     fontupper=\small\raggedright,fontlower=\small\raggedright,
     left=8pt,right=8pt,top=7pt,bottom=6pt,
     before skip=0pt,after skip=0pt,
     title={#1},
     fonttitle=\normalfont\small\bfseries\raggedright,coltitle=white,colbacktitle=black,
     toptitle=3pt,bottomtitle=3pt,
     segmentation style={draw=black!40,dashed,line width=0.35pt}]}
  {\end{tcolorbox}
   \caption{#3}\label{#2}
   \end{minipage}\par\addvspace{7pt}}
